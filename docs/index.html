C✓ बना और चला

#समावेश <stdio.h>
/* क्रमबद्ध सरणी में द्विआधारी खोज */
पूर्णांक द्विआधारी_खोज(पूर्णांक मूल्यावली[], पूर्णांक गिनती, पूर्णांक कुंजी) {
  पूर्णांक निचला = 0, ऊपरी = गिनती - 1;
  जबतक (निचला <= ऊपरी) {
    पूर्णांक मध्य = निचला + (ऊपरी - निचला) / 2;
    अगर (मूल्यावली[मध्य] == कुंजी) वापस मध्य;
    अगर (मूल्यावली[मध्य] < कुंजी) निचला = मध्य + 1; अन्यथा ऊपरी = मध्य - 1;
  }
  वापस -1;
}
पूर्णांक मुख्य(व्योम) {
  पूर्णांक मूल्यावली[] = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
  म_लिखो("%d %d\n", द्विआधारी_खोज(मूल्यावली, 10, 23), द्विआधारी_खोज(मूल्यावली, 10, 7));
  वापस 0;
}

  निजी पूर्णांक शीर्ष = 0;
  खुला व्योम धकेलो(पूर्णांक मूल्य) { वस्तुएँ[शीर्ष++] = मूल्य; }
  खुला पूर्णांक निकालो() { वापस वस्तुएँ[--शीर्ष]; }
  खुला जड़ व्योम मुख्य(माला_श्रेणी[] तर्क) {
    ढेर वस्तुएँ = नया ढेर();
    क्रम (पूर्णांक मूल्य = 1; मूल्य <= 3; मूल्य++) वस्तुएँ.धकेलो(मूल्य * 10);
    प्रणाली.बाहर.म_लिखो_न(वस्तुएँ.निकालो() + " " + वस्तुएँ.निकालो());
